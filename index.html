function simPrice(base,seed){

  const r=(Math.sin(seed+Date.now()/60000)*0.5+0.5);

  return+(base*(1+(r-0.5)*0.02)).toFixed(2);

}

function initStocks(){

  companies.forEach((c,i)=>{

    const p=simPrice(c.stock.price,i);

    const chg=+(p-c.stock.price).toFixed(2);

    stockPrices[c.ticker]={...c.stock,price:p,chg,pct};

  });

}
